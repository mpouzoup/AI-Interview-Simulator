0033     internal static (string Id, string Text) Select(IReadOnlyList<CriterionEvidence> criteria)
0034     {
0035         if (criteria.Any(c => c.Reason == EvidenceReason.ConflictingEvidence))
0036             return ("q3.fallback", Fallback);
0037         if (criteria[0].Reason == EvidenceReason.ExplicitNonAnswer)
0038             return ("q3.example", "Μπορείτε να επιλέξετε ένα μικρό περιστατικό από σπουδές, εργασία ή προσωπικό έργο. Περιγράψτε τη δυσκολία, τι κάνατε εσείς και πώς εξελίχθηκε η κατάσταση.");
0039         var p = criteria[0].State == CriterionState.Detected;
0040         var a = criteria[1].State == CriterionState.Detected;
0041         var r = criteria[2].State == CriterionState.Detected;
0042         if (p && a && r)
0043             return ("q3.complete", "Στην απάντησή σας αναγνωρίζονται στοιχεία για τη δυσκολία, τις προσωπικές σας ενέργειες και την έκβαση. Μπορείτε να κάνετε ακόμη πιο σαφή τη σύνδεση ανάμεσα στις ενέργειές σας και στην εξέλιξη της κατάστασης.");
0044         if (p && a)
0045             return ("q3.result", "Η απάντησή σας περιγράφει τη δυσκολία και προσωπικές σας ενέργειες. Για να γίνει ακόμη πιο σαφής, μπορείτε να εξηγήσετε και ποια ήταν η τελική έκβαση, ακόμη και αν δεν ήταν η επιθυμητή.");
0046         if (p && !a)
0047             return ("q3.action", "Η απάντησή σας δίνει στοιχεία για τη δυσκολία που αντιμετωπίσατε. Μπορείτε να διευκρινίσετε ποια συγκεκριμένα βήματα κάνατε εσείς για να την αντιμετωπίσετε.");
0048         if (a && r)
0049             return ("q3.problem", "Η απάντησή σας δίνει στοιχεία για προσωπικές ενέργειες και έκβαση. Μπορείτε να κάνετε πιο συγκεκριμένο το αρχικό πρόβλημα και το πλαίσιο στο οποίο προέκυψε.");
0050         return ("q3.fallback", Fallback);
0051     }

0175     internal static (string Id, string Text) Select(int stageNumber, Stage stage, IReadOnlyList<CriterionEvidence> criteria)
0176     {
0177         var prefix = $"q{stageNumber}.";
0178         if (criteria.Any(c => c.Reason == EvidenceReason.ConflictingEvidence) ||
0179             !criteria.Any(c => c.State == CriterionState.Detected))
0180             return (prefix + "fallback", stage.Fallback);
0181         for (var i = 0; i < stage.Criteria.Length; i++)
0182             if (stage.Criteria[i].Required && criteria[i].State != CriterionState.Detected)
0183                 return (prefix + stage.Criteria[i].Id, stage.Criteria[i].Prompt);
0184         if (stage.Criteria.Where((c, i) => !c.Required && criteria[i].State != CriterionState.Detected).Any())
0185             return (prefix + "enrich", stage.Enrichment);
0186         return (prefix + "complete", stage.Complete);
0187     }
