0023     private async Task<IActionResult> CurrentFlowState(InterviewSession session)
0024     {
0025         if (await HasPendingQuestionnaires(session.Id, "pre"))
0026             return Ok(new { flowVersion = 2, flowState = "PreQuestionnaires", stageNumber = 1, isFinished = false });
0027         var answers = session.Answers.OrderBy(a => a.StageNumber).ToList();
0028         var last = answers.LastOrDefault();
0029         if (last != null && last.StageCompletedAt == null)
0030         {
0031             if (last.ChoseToRevise == true && last.RevisedAnswerText == null)
0032                 return FlowReply(last, 1, "Revising");
0033             var round = last.RevisedAnswerText != null ? 2 : 1;
0034             var text = round == 1 ? last.FeedbackText : last.RevisedFeedbackText;
0035             return FlowReply(last, round, FeedbackResult.IsUsable(text)
0036                 ? round == 1 ? "WaitingForDecision" : "WaitingForContinue"
0037                 : round == 1 ? "AwaitingFeedback" : "AwaitingRevisedFeedback", "NotRequired");
0038         }
0039         if (session.FinishedAt != null && await HasPendingQuestionnaires(session.Id, "post"))
0040             return Ok(new { flowVersion = 2, flowState = "PostQuestionnaires", stageNumber = 7, isFinished = false });
0041         if (session.FinishedAt != null)
0042             return Ok(new { flowVersion = 2, flowState = "Finished", isFinished = true });
0043         var stage = answers.Count + 1;
0044         if (stage > 7) return Conflict("Invalid incomplete session state.");
0045         return Ok(new
0046         {
0047             flowVersion = 2, flowState = "Answering", stageNumber = stage,
0048             questionText = await _interviewManager.GetNextQuestionAsync(session.Condition, stage, answers),
0049             feedbackStatus = "NotRequired", isFinished = false
0050         });
0051     }
