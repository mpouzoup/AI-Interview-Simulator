0023     public async Task<FeedbackResult> GenerateResponseAsync(string prompt, CancellationToken cancellationToken = default)
0024     {
0025         if (string.IsNullOrWhiteSpace(_apiKey))
0026             return new(FeedbackStatus.Unavailable, ErrorCode: "missing_api_key");
0027 
0028         using var budget = new CancellationTokenSource(TotalTimeout, _timeProvider);
0029         using var linked = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken, budget.Token);

0040                     }
0041                 }

0052             "gemini-3.6-flash",

0054             "gemini-3.7-flash",
0055             "gemini-flash-latest"
0056         };
0057

0059         string? currentModel = null;

0061         try
0062         {
0063             foreach (var model in candidateModels)
0064             {
0065                 currentModel = model;
0066                 var maxAttempts = model == "gemini-3.6-flash" ? 2 : 1;
0067                 for (var attempt = 1; attempt <= maxAttempts; attempt++)
0068                 {
0069                     token.ThrowIfCancellationRequested();
0070                     attempts++;
0071                     using var request = new HttpRequestMessage(HttpMethod.Post,
0072                         $"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent");
0073                     request.Headers.Add("x-goog-api-key", _apiKey);
0074                     request.Content = new StringContent(JsonSerializer.Serialize(requestBody), Encoding.UTF8, "application/json");
0075                     var retryable = true;
0076                     try
0077                     {

0079                         if (response.IsSuccessStatusCode)

0081                             var json = await response.Content.ReadAsStringAsync(token);
0082                             var parsed = ParseResponse(json) with { Model = model, Attempts = attempts };
0083                             // A structurally invalid/empty response is not a transport retry.
0084                             return parsed;
0085                         }
0086                         lastFailure = new(FeedbackStatus.Unavailable, ErrorCode: $"http_{(int)response.StatusCode}");
0087                         retryable = response.StatusCode is HttpStatusCode.RequestTimeout or HttpStatusCode.TooManyRequests
0088                             or HttpStatusCode.InternalServerError or HttpStatusCode.BadGateway
0089                             or HttpStatusCode.ServiceUnavailable or HttpStatusCode.GatewayTimeout;

0096                     {
0097                         // A shorter HttpClient timeout must not start another long request.
0098                         return new(FeedbackStatus.Unavailable, ErrorCode: "timeout", Model: model, Attempts: attempts);
0099                     }

0102                         await Task.Delay(TimeSpan.FromSeconds(2), _timeProvider, token);

0112         return lastFailure with { Model = currentModel, Attempts = attempts };
