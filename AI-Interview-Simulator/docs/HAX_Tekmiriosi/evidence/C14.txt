0462     {
0463         var answer = await _context.UserAnswers
0464             .FirstOrDefaultAsync(a => a.Id == answerId);

0473             return BadRequest("Revised answer cannot be empty.");

0486                 : Conflict("This answer has already been revised with different text.");
0487         }
0488

0491         {

0493             if (!await IsActiveAnswer(answer) || answer.FeedbackDisplayAcknowledgedAt == null)
0494                 return Conflict("Revision is not allowed in the current state.");
0495             answer.RevisionBrowserSubmittedAt = request.Timing?.BrowserOccurredAt;
0496             answer.RevisionWritingMilliseconds = request.Timing?.ElapsedMilliseconds;
0497         }
0498 
0499         answer.RevisedAnswerText = request.RevisedAnswerText.Trim();

0502         if (answer.StageNumber >= 7 && revisionSession?.FlowVersion < 2)
0503         {
0504             var session = await _context.InterviewSessions

0509             }
0510         }
